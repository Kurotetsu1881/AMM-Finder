(async () => {
  // ===== AYARLAR =====
  const CHAPTERS = 'ALL';    // tüm chapter'lar. Sadece bazıları için: ['21','24','25']
  const DELAY = 150;         // istekler arası bekleme (ms)
  // ===================
  const REF = /^(\d{2}-\d{2}-\d{2}-\d{3}-\d{3})(?:-([A-Z0-9]+))?\s*-\s*(.+)$/;
  const sleep = ms => new Promise(r => setTimeout(r, ms));

  // Adresten otomatik: portal yolu ve doküman öneki (A330, A350... her tip için farklıdır)
  const seg = location.pathname.split('/')[1];
  const PREFIX = (location.pathname.match(/\/document\/([^\/]+)\//) || [])[1];
  if (!PREFIX) return console.error('AMM sayfasında değilsiniz (adreste /document/<kod>/ yok).');
  const BASE = `${location.origin}/${seg}/api/toc/${PREFIX}/getTocItems?id=`;

  // Uçak tipi
  const wc = new URL(location.href).searchParams.get('wc') || '';
  const type = ((/actype:([^;]+)/i.exec(wc) || /AMM - ([A-Z0-9]+) - /i.exec(document.title) || [])[1] || 'UNKNOWN').toUpperCase();

  // Revizyon numarası ve tarihi: sayfa başlığından ("... Rev. num: 84 - 01-Oct-2026 ...")
  const rm = /Rev\.?\s*num:?\s*(\d+)\s*-\s*(\d{1,2}-[A-Za-z]{3}-\d{4})/i.exec(document.title);
  let revNum = rm ? rm[1] : null, revDate = rm ? rm[2] : null;
  if (!revNum) {
    revNum = (prompt('Revizyon numarası bulunamadı. Kapak sayfasındaki "Revision number" değerini yazın:') || '').trim();
    revDate = (prompt('Revizyon tarihi (örn. 01-Oct-2026):') || '').trim();
    if (!revNum) return console.error('Revizyon bilgisi olmadan tarama iptal edildi.');
  }
  console.log(`${type} AMM · REV:${revNum} · ${revDate} · önek: ${PREFIX}`);

  // "Aç" linki şablonu: o an açık sayfanın adresi, itemId yerine {ID}
  const pageUrl = new URL(location.href);
  if (!/_EN\d{12,}$/.test(pageUrl.searchParams.get('itemId') || '')) {
    console.warn('Uyarı: açık sayfa bir görev (task) sayfası değil. "Aç" linkleri çalışmayabilir. Önce sol ağaçtan bir görev açıp kodu tekrar çalıştırın.');
  }
  pageUrl.searchParams.set('itemId', '{ID}');
  const linkTemplate = pageUrl.toString().replace('%7BID%7D', '{ID}');

  // Effectivity ayrıştırma: "** ON A/C FSN 201-250, 366, 401-450"
  const parseApp = a => {
    const raw = String(a || '').replace(/\s+/g, ' ').trim();
    const body = raw.replace(/^[*\s]*ON A\/C\s*(?:FSN\s*)?/i, '').trim();
    if (!body || /^ALL$/i.test(body)) return {app: raw, fsn: null, unk: false};
    const ranges = []; let unk = false;
    body.split(/[,;]/).forEach(tok => {
      tok = tok.trim(); if (!tok) return;
      let m;
      if ((m = /^(\d+)\s*-\s*(\d+)$/.exec(tok))) ranges.push([+m[1], +m[2]]);
      else if ((m = /^(\d+)$/.exec(tok))) ranges.push([+m[1], +m[1]]);
      else unk = true;
    });
    return {app: raw, fsn: !unk && ranges.length ? ranges : null, unk: unk || !ranges.length};
  };

  const list = CHAPTERS === 'ALL'
    ? Array.from({length: 76}, (_, i) => String(i + 5).padStart(2, '0'))
    : CHAPTERS;
  const tasks = new Map();
  const okCh = [], skipCh = [], badCh = [], todoCh = [];
  let n = 0, skipped = 0;
  window.__stop = false;

  const addSol = (s, into) => {
    if (!s) return;
    const m = REF.exec((s.title || '').trim());
    if (m && s.id) {
      if (!into.has(s.id) && m[3].trim().toUpperCase() !== 'DELETED') {
        const a = parseApp(s.applicability);
        into.set(s.id, {id: s.id, ch: m[1].slice(0, 2), ref: m[1], rev: m[2] || '',
                        title: m[3].trim(), app: a.app, fsn: a.fsn, unk: a.unk});
      }
    } else if (s.title) skipped++;
    (s.solutions || []).forEach(x => addSol(x, into));
  };

  const getJson = async (id, tries) => {
    for (let k = 0; k < tries; k++) {
      try {
        const r = await fetch(BASE + encodeURIComponent(id), {credentials: 'include'});
        if (r.status === 404) return {notFound: true};
        if (!r.ok) throw new Error(r.status);
        return await r.json();
      } catch (e) { await sleep(500 * (k + 1)); }
    }
    return null;
  };

  // Chapter chapter, derinlemesine gez: her chapter bitince "tamam" yazar
  for (const ch of list) {
    if (window.__stop) { todoCh.push(ch); continue; }
    const rootId = `${PREFIX}_EN${ch}`;
    const root = await getJson(rootId, 2);
    if (!root || root.notFound || !((root.childTocItems || []).length || (root.listSolutions || []).length)) {
      skipCh.push(ch); continue;
    }
    const mine = new Map(), seen = new Set([rootId]), stack = [], failed = [];
    let stopped = false;
    const visit = j => {
      (j.childTocItems || []).forEach(c => { if (c.id && !seen.has(c.id)) { seen.add(c.id); stack.push(c.id); } });
      (j.listSolutions || []).forEach(x => addSol(x, mine));
    };
    visit(root);
    for (let pass = 0; pass < 2 && !stopped; pass++) {
      if (pass === 1) {                       // başarısız düğümleri bir kez daha dene
        stack.push(...failed.splice(0));
        if (!stack.length) break;
        await sleep(3000);
      }
      while (stack.length) {
        if (window.__stop) { stopped = true; break; }
        const id = stack.pop();
        const j = await getJson(id, 3);
        if (!j || j.notFound) failed.push(id); else visit(j);
        if (++n % 50 === 0) console.log(`Chapter ${ch} | ${n} istek | yığın ${stack.length} | ${mine.size} görev`);
        await sleep(DELAY);
      }
    }
    if (stopped) { todoCh.push(ch); continue; }
    if (failed.length) { badCh.push(ch); console.warn(`Chapter ${ch} EKSİK kaldı (${failed.length} düğüm alınamadı), dosyaya girmeyecek.`); continue; }
    mine.forEach((v, k) => tasks.set(k, v));
    okCh.push(ch);
    console.log(`✔ Chapter ${ch} tamam: ${mine.size} görev`);
  }

  if (!okCh.length) return console.error('Tamamlanan chapter yok, dosya oluşturulmadı.');
  const rows = [...tasks.values()].sort((a, b) => a.ref.localeCompare(b.ref) || a.rev.localeCompare(b.rev));
  const out = {meta: {type, revNum, revDate, prefix: PREFIX, linkTemplate, pageTitle: document.title,
                      exported: new Date().toISOString(), chapters: okCh, count: rows.length}, tasks: rows};
  const a = document.createElement('a');
  a.href = URL.createObjectURL(new Blob([JSON.stringify(out)], {type: 'application/json'}));
  a.download = `amm_${type}_REV${revNum}_${okCh.length > 6 ? 'ALL' : okCh.join('-')}.json`;
  a.click();
  window.__tasks = tasks;
  const redo = [...badCh, ...todoCh];
  console.log(`BİTTİ: ${rows.length} görev, ${n} istek, ${skipped} atlanan kayıt. Tamamlanan chapter: ${okCh.join(',')}`);
  if (redo.length) console.warn(`Tamamlanmayanlar için kodu tekrar çalıştırın: const CHAPTERS = ${JSON.stringify(redo)};`);
})();
